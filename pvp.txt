local CONFIG_FILE = "migpvp_config.json"

local function saveConfig()
    if not writefile then return end
    pcall(function()
        writefile(CONFIG_FILE, HttpService:JSONEncode(CONFIG))
    end)
end

local function loadConfig()
    if not isfile or not readfile then return end
    if not isfile(CONFIG_FILE) then return end
    local ok, content = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not content then return end
    local decOk, data = pcall(function() return HttpService:JSONDecode(content) end)
    if not decOk or type(data) ~= "table" then return end
    for k, v in pairs(data) do
        if CONFIG[k] ~= nil then CONFIG[k] = v end
    end
    print("[CONFIG] Config carregada")
end

-- ============ HELPERS ============
local function isAlly(player)
    if not player or not player.Team or not LP.Team then return false end
    return player.Team == LP.Team
end

local function isEnemy(player)
    if not player or player == LP then return false end
    if CONFIG.aimTeamCheck and isAlly(player) then return false end
    return true
end

local function isPlayerBlacklisted(player)
    if not player then return true end
    local bl = CONFIG.playerBlacklist
    return bl[player.UserId] == true
        or bl[tostring(player.UserId)] == true
        or bl[player.Name] == true
        or bl[string.lower(player.Name)] == true
end

local function inSafeZone(player)
    if not player or not player.Character then return false end
    local char = player.Character
    if char:FindFirstChildOfClass("ForceField") then return true end
    if char:GetAttribute("SafeZone") == true then return true end
    if player:GetAttribute("PvpDisabled") == true then return true end
    local wo = Workspace:FindFirstChild("_WorldOrigin")
    local zones = wo and wo:FindFirstChild("SafeZones")
    if zones then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then
            for _, zone in ipairs(zones:GetChildren()) do
                if zone:IsA("BasePart") then
                    local radius = math.min(zone.Size.X, zone.Size.Z) / 2
                    if radius <= 0 then radius = 350 end
                    if radius > 500 then radius = 350 end
                    if (zone.Position - hrp.Position).Magnitude <= radius then return true end
                end
            end
        end
    end
    return false
end

local function isTargetValid(player, maxDist)
    if not player or player == LP then return false end
    if isPlayerBlacklisted(player) then return false end
    if not player.Character then return false end
    local char = player.Character
    if not char:IsDescendantOf(Workspace) then return false end
    if not isEnemy(player) then return false end
    if CONFIG.aimSafeZoneCheck and inSafeZone(player) then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart
    if not hum or hum.Health <= 0 or not hrp then return false end
    if hrp.Position.Y <= -500 then return false end
    local myChar = LP.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHrp then return false end
    local dist = (hrp.Position - myHrp.Position).Magnitude
    if dist > (maxDist or CONFIG.aimMaxDistance) then return false end
    return true, dist
end

-- ============ AIM SCORE ============
local function aimCenter()
    if CONFIG.aimFovOrigin == "Mouse" and not UserInputService.TouchEnabled then
        local p = UserInputService:GetMouseLocation()
        if p.X > 0 and p.Y > 0 then return p end
    end
    return Camera.ViewportSize / 2
end

local function aimScore(part, distance)
    if CONFIG.aimMode == "360" then return distance end
    if not part then return nil end
    if CONFIG.aimMode == "180" then
        local delta = part.Position - Camera.CFrame.Position
        if delta:Dot(Camera.CFrame.LookVector) < 0 then return nil end
        return distance
    end
    local point, visible = Camera:WorldToViewportPoint(part.Position)
    if not visible or point.Z <= 0 then return nil end
    local radius = math.clamp(tonumber(CONFIG.aimFovRadius) or 300, 20, 800)
    local pixels = (Vector2.new(point.X, point.Y) - aimCenter()).Magnitude
    return pixels <= radius and pixels or nil
end

-- ============ SELECT TARGET ============
local function selectTarget()
    local best, bestScore, bestModel, bestPart = nil, math.huge, nil, nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP then
            local ok, valid, distance = pcall(isTargetValid, p, CONFIG.aimMaxDistance)
            if ok and valid then
                local char = p.Character
                local part = char:FindFirstChild(CONFIG.aimTargetPart)
                          or char:FindFirstChild("HumanoidRootPart")
                if part then
                    local score = aimScore(part, distance)
                    if score and score < bestScore then
                        best, bestScore, bestModel, bestPart = p, score, char, part
                    end
                end
            end
        end
    end
    return best, bestModel, bestPart
end

-- ============ UPDATE TARGET ============
local function updateTarget()
    if not CONFIG.aimEnabled then
        CurrentTarget, CurrentTargetChar, CurrentTargetPart = nil, nil, nil
        return
    end
    local p, char, part = selectTarget()
    CurrentTarget, CurrentTargetChar, CurrentTargetPart = p, char, part
end

-- ============ QUERY ============
local function queryAimPart()
    if not CONFIG.aimEnabled then return nil end
    if not CurrentTargetPart or not CurrentTargetPart.Parent then return nil end
    return CurrentTargetPart, CurrentTargetPart.Position
end

-- ============ READ AIM ============
local function readAimProperty(key)
    if key ~= "Hit" and key ~= "Target" and key ~= "UnitRay" then return false end
    local part, position = queryAimPart()
    if not part then return false end
    if key == "Hit" then return true, CFrame.new(position) end
    if key == "Target" then return true, part end
    local cam = Workspace.CurrentCamera
    if not cam then return false end
    local dir = position - cam.CFrame.Position
    if dir.Magnitude < 0.001 then return false end
    return true, Ray.new(cam.CFrame.Position, dir.Unit)
end

-- ============ HOOK 1: rawMouse ============
task.spawn(function()
    pcall(function()
        local env = getgenv()
        local wrap = type(newcclosure) == "function" and newcclosure or function(fn) return fn end
        env._MigActiveIndexHandler = nil
        if not env._MigOldIndex then
            local function dispatcher(object, key)
                local handler = env._MigActiveIndexHandler
                if handler then return handler(object, key) end
                local native = env._MigOldIndex
                if type(native) == "function" then return native(object, key) end
                if type(native) == "table" then return native[key] end
            end
            if type(hookmetamethod) == "function" then
                env._MigOldIndex = hookmetamethod(rawMouse, "__index", wrap(dispatcher))
            elseif type(getrawmetatable) == "function" and type(setreadonly) == "function" then
                local mt = getrawmetatable(rawMouse)
                env._MigOldIndex = mt.__index
                setreadonly(mt, false); mt.__index = wrap(dispatcher); setreadonly(mt, true)
            end
        end
        local native = env._MigOldIndex
        if type(native) == "function" or type(native) == "table" then
            env._MigActiveIndexHandler = function(object, key)
                if object == rawMouse and CONFIG.aimEnabled
                   and not (checkcaller and checkcaller()) then
                    local changed, value = readAimProperty(key)
                    if changed then return value end
                end
                if type(native) == "function" then return native(object, key) end
                return native[key]
            end
        end
    end)
end)

-- ============ HOOK 2: modulo Mouse ============
task.spawn(function()
    pcall(function()
        local instance = ReplicatedStorage:WaitForChild("Mouse", 10)
        if not instance then return end
        local object = require(instance)
        if type(object) ~= "table" then return end
        local mt = type(getrawmetatable) == "function"
            and getrawmetatable(object) or getmetatable(object)
        if mt ~= nil and type(mt) ~= "table" then return end
        if not mt then mt = {}; setmetatable(object, mt) end
        local oldIndex = mt.__index
        local oldNewIndex = mt.__newindex
        local values = { Hit = rawget(object, "Hit"), Target = rawget(object, "Target") }
        local stored = { Hit = values.Hit ~= nil, Target = values.Target ~= nil }
        local wasReadOnly = type(isreadonly) == "function" and isreadonly(mt) or false
        if type(setreadonly) == "function" then setreadonly(mt, false) end
        local function originalIndex(self, key)
            if stored[key] then return values[key] end
            if type(oldIndex) == "function" then return oldIndex(self, key) end
            if type(oldIndex) == "table" then return oldIndex[key] end
            return nil
        end
        local ownIndex = function(self, key)
            if CONFIG.aimEnabled and (key == "Hit" or key == "Target") then
                local changed, value = readAimProperty(key)
                if changed then return value end
            end
            return originalIndex(self, key)
        end
        local ownNewIndex = function(self, key, value)
            if key == "Hit" or key == "Target" then
                values[key], stored[key] = value, true
            elseif type(oldNewIndex) == "function" then
                oldNewIndex(self, key, value)
            elseif type(oldNewIndex) == "table" then
                oldNewIndex[key] = value
            else
                rawset(self, key, value)
            end
        end
        mt.__index, mt.__newindex = ownIndex, ownNewIndex
        rawset(object, "Hit", nil); rawset(object, "Target", nil)
        if type(setreadonly) == "function" then setreadonly(mt, wasReadOnly) end
        _G._MigRestoreMouseAim = function()
            if mt.__index ~= ownIndex then return end
            if type(setreadonly) == "function" then setreadonly(mt, false) end
            mt.__index = oldIndex
            if mt.__newindex == ownNewIndex then mt.__newindex = oldNewIndex end
            rawset(object, "Hit", values.Hit)
            rawset(object, "Target", values.Target)
            if type(setreadonly) == "function" then setreadonly(mt, wasReadOnly) end
        end
    end)
end)

        task.wait(0.03)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.R, false, game)
    end)
end

local function stopTween()
    tweenActive = false
end

local function startTween()
    tweenActive = true
end

local function toggleTween()
    if tweenActive then stopTween() else startTween() end
end

    "MigTweenToggle",
    function(actionName, inputState)
        if inputState == Enum.UserInputState.Begin then
            toggleTween()
        end

    end,

connect(RunService.Heartbeat, function(dt)
    if not tweenActive then return end
    if not CurrentTargetPart or not CurrentTargetPart.Parent then return end
    local myChar = LP.Character
    if not myChar then return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local targetPos = CurrentTargetPart.Position
    local myPos = myRoot.Position
    local dist = (targetPos - myPos).Magnitude
    local dir = (myPos - targetPos)
    if dir.Magnitude < 0.1 then dir = Vector3.new(0, 0, 1) else dir = dir.Unit end
    local followPos = targetPos + (dir * CONFIG.tweenDistance)
    if dist <= CONFIG.tweenDirectRange then
        pcall(function() myRoot.CFrame = CFrame.new(followPos) end)
    else
        local toTarget = (followPos - myPos)
        local distanceToCover = toTarget.Magnitude
        if distanceToCover > 0.1 then
            local speed = math.clamp(distanceToCover * 3, CONFIG.tweenSpeed, CONFIG.tweenSpeed * 3)
            local step = math.min(speed * dt, distanceToCover)
            local newPos = myPos + (toTarget.Unit * step)
            pcall(function() myRoot.CFrame = CFrame.new(newPos) end)
        end
    end
end)

local function updateHL()
    if highlightObj then highlightObj:Destroy(); highlightObj = nil end
    if CONFIG.highlightAim and CurrentTargetChar and CurrentTargetChar.Parent then
        highlightObj = Instance.new("Highlight")
        highlightObj.FillColor = COLOR_TARGET
        highlightObj.OutlineColor = Color3.fromRGB(255, 255, 255)
        highlightObj.FillTransparency = 0.5
        highlightObj.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlightObj.Parent = CurrentTargetChar
    end
end

local function removeESP(ch)
    local o = espObjects[ch]
    if not o then return end
    for _, v in pairs(o) do pcall(function() v:Destroy() end) end
    espObjects[ch] = nil
end

local function createESP(ch)
    if destroyed or not CONFIG.espEnabled then return end
    if not ch or ch == LP.Character then return end
    removeESP(ch)
    local root = ch:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local hl = Instance.new("Highlight")
    hl.FillColor = COLOR_OTHER
    hl.FillTransparency = 0.5
    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = ch
    local bb = Instance.new("BillboardGui")
    bb.Adornee = root
    bb.Size = UDim2.fromOffset(100, 18)
    bb.StudsOffset = Vector3.new(0, 3.5, 0)
    bb.AlwaysOnTop = true
    bb.MaxDistance = 5000
    bb.Parent = root
    local dl = Instance.new("TextLabel")
    dl.Size = UDim2.new(1, 0, 1, 0)
    dl.BackgroundTransparency = 1
    dl.Text = "0"
    dl.TextColor3 = COLOR_OTHER
    dl.TextStrokeColor3 = Color3.new(0, 0, 0)
    dl.TextStrokeTransparency = 0
    dl.TextSize = 13
    dl.Font = Enum.Font.GothamBold
    dl.Parent = bb
    espObjects[ch] = { highlight = hl, billboard = bb, distance = dl }
end

local function setupPlayer(p)
    if p == LP then return end
    if p.Character then task.defer(function() createESP(p.Character) end) end
    connect(p.CharacterAdded, function(ch)
        task.wait(0.3)
        if not destroyed and CONFIG.espEnabled then createESP(ch) end
    end)
end

for _, p in ipairs(Players:GetPlayers()) do setupPlayer(p) end

-- ============ MAIN LOOP ============
local espTimer = 0
connect(RunService.RenderStepped, function(dt)
    if CONFIG.aimEnabled then
        local oldPart = CurrentTargetPart
        updateTarget()
        if CurrentTargetPart ~= oldPart then updateHL() end
    end
    espTimer = espTimer + dt
    if espTimer >= 0.15 then
        espTimer = 0
        if not destroyed and CONFIG.espEnabled then
            local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            if myRoot then
                for ch, o in pairs(espObjects) do
                    if not ch or not ch.Parent then
                        removeESP(ch)
                    else
                        local r = ch:FindFirstChild("HumanoidRootPart")
                        if r and o.distance then
                            o.distance.Text = math.floor((myRoot.Position - r.Position).Magnitude)
                        end
                    end
                end
            end
        end
    end
end)

-- ============ SPEED ============
connect(RunService.RenderStepped, function()
    if destroyed or not CONFIG.speedEnabled then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    local d = hum.MoveDirection
    if d.Magnitude > 0.1 then
        root.CFrame = root.CFrame + d.Unit * ((CONFIG.speedValue - 16) / 60)
    end
end)

-- ============ DASH ============
connect(UserInputService.InputBegan, function(input, gpe)
    if destroyed or not CONFIG.dashEnabled or gpe then return end
    if input.KeyCode ~= Enum.KeyCode.Q then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    local d = hum.MoveDirection
    if d.Magnitude < 0.1 then d = root.CFrame.LookVector end
    for i = 1, 2 do
        if destroyed or not CONFIG.dashEnabled then break end
        root.CFrame = root.CFrame + d.Unit * CONFIG.dashPower
        task.wait(CONFIG.dashDelay)
    end
end)

-- ============ TP (G) ============
connect(UserInputService.InputBegan, function(input, gpe)
    if gpe then return end
    if input.KeyCode == CONFIG.tpKey then
        local myChar = LP.Character
        if not myChar then return end
        local myHrp = myChar:FindFirstChild("HumanoidRootPart")
        if not myHrp then return end
        if CurrentTargetPart and CurrentTargetPart.Parent then
            local targetPos = CurrentTargetPart.Position
            local dist = (targetPos - myHrp.Position).Magnitude
            if dist > CONFIG.tpMaxDistance then
                targetPos = myHrp.Position + (targetPos - myHrp.Position).Unit * CONFIG.tpMaxDistance
            end
            myHrp.CFrame = CFrame.new(targetPos)
        end
    end
end)

    end

_G._UnloadMigAim = function()
    if destroyed then return end
    destroyed = true
    stopTween()
    pcall(function() ContextActionService:UnbindAction("MigSoruAim") end)
    pcall(function() ContextActionService:UnbindAction("MigTweenToggle") end)
    for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
    table.clear(connections)
    for ch in pairs(espObjects) do removeESP(ch) end
    table.clear(espObjects)
    if highlightObj then highlightObj:Destroy() end
    if _G._MigRestoreMouseAim then pcall(_G._MigRestoreMouseAim) end
    pcall(function() gui:Destroy() end)
    pcall(function() buttonsGui:Destroy() end)
    _G._UnloadMigAim = nil
    print("[MIG] Descarregado.")
end
